    Function connectAgent

    • connectAgent(options: ConnectAgentOptions): Promise<AgentRuntimeHandle>

      Connects to an AI agent through a transport layer, providing a unified interface for agent interaction.

      This is the primary entry point for the AgentKit SDK. It handles agent and transport resolution, initialization, compatibility validation, and returns a runtime handle for interaction.

      Parameters

      • options: ConnectAgentOptions

        Configuration options for the agent connection

        Options for connecting to an agent with plugin verification support

          Allow unsigned plugins in development mode

      Returns Promise<AgentRuntimeHandle>

      Promise resolving to an AgentRuntimeHandle for interacting with the connected agent

      Example

      import { connectAgent, AgentRegistry } from '@agentkit/sdk'
      import { DummyAgent } from '@agentkit/test-agent'
      import { DummyTransport } from '@agentkit/transports-dummy'

      // Register components
      AgentRegistry.register({
        id: 'my-agent',
        name: 'My Agent',
        version: '1.0.0',
        factory: () => new DummyAgent()
      })
      AgentRegistry.registerTransport(new DummyTransport())

      // Connect and use
      const handle = await connectAgent({
        agent: 'my-agent',
        transport: 'dummy',
        onOutput: (msg) => console.log('Response:', msg)
      })

      await handle.send({ message: 'Hello!' })
      await handle.shutdown()

      Example: Production usage with observability:

      const handle = await connectAgent({
        agent: 'production-agent',
        transport: 'http',
        resourceLimits: {
          maxConcurrentRequests: 10,
          maxQueueSize: 50,
          requestTimeoutMs: 30000
        },
        circuitBreaker: {
          failureThreshold: 5,
          timeout: 60000
        },
        observability: {
          onRequestStart: (metadata) => console.log('Request started:', metadata.requestId),
          onRequestEnd: (metadata, duration) => console.log('Request completed in', duration, 'ms'),
          onCircuitBreakerStateChange: (event) => console.log('Circuit breaker:', event.toState)
        }
      })
